#include <Wire.h>
#include <I2CKeyPad.h>
#include <LiquidCrystal_I2C.h>
#include <Adafruit_Fingerprint.h>
#include <HardwareSerial.h>


#define KEYPAD_ADDR 0x20
#define LCD_ADDR 0x27


I2CKeyPad keypad(KEYPAD_ADDR);
LiquidCrystal_I2C lcd(LCD_ADDR, 16, 2);


// 🔥 Fingerprint setup (ADDED)
HardwareSerial mySerial(2);
Adafruit_Fingerprint finger = Adafruit_Fingerprint(&mySerial);


// LED pins
int redLED = 3;
int greenLED = 5;


// IR sensor
int irPin = 4;


// buzzer
int buzzerPin = 18;


// keypad mapping
char keymap[16] = {
  'D', '#', '0', '*',
  'C', '9', '8', '7',
  'B', '6', '5', '4',
  'A', '3', '2', '1'
};


uint8_t lastIndex = 0xFF;


// password system
String correctCode = "1234";
String secretCode = "9999";
String input = "";
int wrongAttempts = 0;


bool isDisarmed = false;
bool alarmTriggered = false;
bool secretMode = false;
String alertType = "";


void setup() {


  delay(1000);


  Serial.begin(115200);


  Wire.begin(21, 22);
  keypad.begin();


  lcd.init();
  lcd.backlight();


  pinMode(redLED, OUTPUT);
  pinMode(greenLED, OUTPUT);
  pinMode(buzzerPin, OUTPUT);
  pinMode(irPin, INPUT_PULLUP);


  digitalWrite(redLED, HIGH);
  digitalWrite(greenLED, LOW);


  lcd.setCursor(0,0);
  lcd.print("ENTER CODE:");


  // 🔥 Fingerprint init (ADDED)
  mySerial.begin(57600, SERIAL_8N1, 16, 17);
  finger.begin(57600);


  if (finger.verifyPassword()) {
    Serial.println("Fingerprint READY");
  } else {
    Serial.println("Fingerprint NOT FOUND");
  }
}


void loop() {


  // 🔥 FINGERPRINT CHECK (ADDED — DOES NOT BREAK ANYTHING)
  if (!isDisarmed) {
    if (finger.getImage() == FINGERPRINT_OK) {
      if (finger.image2Tz() == FINGERPRINT_OK) {
        if (finger.fingerSearch() == FINGERPRINT_OK) {


          // ✅ MATCH → DISARM
          isDisarmed = true;
          alarmTriggered = false;
          wrongAttempts = 0;
          input = "";
          secretMode = false;


          noTone(buzzerPin);


          digitalWrite(redLED, LOW);
          digitalWrite(greenLED, HIGH);


          lcd.clear();
          lcd.setCursor(0,0);
          lcd.print("FIngerprint success");
          lcd.setCursor(0,1);
          lcd.print("Alarm has been disarmed");
          tone(buzzerPin, 1200);
          delay(150);
          noTone(buzzerPin);
          delay(100);
          tone(buzzerPin, 1500);
          delay(150);
          noTone(buzzerPin);


          delay(1000);
          lcd.clear();
        }
        else
        {
          lcd.clear();
          lcd.setCursor(0,0);
          lcd.print("NO MATCH");


          tone(buzzerPin, 400);
          delay(200);
          noTone(buzzerPin);


          delay(800);
          lcd.clear();
          lcd.setCursor(0,0);
          lcd.print("ENTER CODE:");
        }
      }
    }
  }


  // ================= DISARMED MODE =================
  if (isDisarmed) {


    digitalWrite(redLED, LOW);
    digitalWrite(greenLED, HIGH);
    noTone(buzzerPin);


    lcd.setCursor(0,0);
    lcd.print("SYSTEM DISARMED ");
    lcd.setCursor(0,1);
    lcd.print("Press A to ARM  ");


    uint8_t index = keypad.getKey();


    if (index != 0xFF) {
      if (index < 16) {
        char key = keymap[index];


        if (key == 'A') {
          isDisarmed = false;
          alarmTriggered = false;
          wrongAttempts = 0;
          input = "";
          secretMode = false;


          lcd.clear();
          lcd.setCursor(0,0);
          lcd.print("SYSTEM ARMED");


          digitalWrite(redLED, HIGH);
          digitalWrite(greenLED, LOW);


          tone(buzzerPin, 800);
          delay(150);
          noTone(buzzerPin);


          delay(800);


          lcd.clear();
          lcd.setCursor(0,0);
          lcd.print("ENTER CODE:");
        }
      }


      lastIndex = index;
      delay(150);
    }


    if (index == 0xFF) {
      lastIndex = 0xFF;
    }


    return;
  }


  // ================= MOTION DETECTION =================
  if (digitalRead(irPin) == LOW) {
    alarmTriggered = true;
    alertType = "MOTION";
  }


  // ================= KEYPAD =================
  uint8_t index = keypad.getKey();


  if (index != 0xFF && index != lastIndex && !alarmTriggered) {


    if (index < 16) {
      char key = keymap[index];


      if (key == '#') {
        lcd.clear();


        if (input == correctCode) {


          isDisarmed = true;
          wrongAttempts = 0;
          secretMode = false;


          lcd.setCursor(0,0);
          lcd.print("DISARMED");


          digitalWrite(redLED, LOW);
          digitalWrite(greenLED, HIGH);


          tone(buzzerPin, 1200);
          delay(150);
          noTone(buzzerPin);
          delay(100);
          tone(buzzerPin, 1500);
          delay(150);
          noTone(buzzerPin);
        }
        else {
          wrongAttempts++;


          lcd.setCursor(0,0);
          lcd.print("WRONG PASS");


          digitalWrite(redLED, HIGH);
          digitalWrite(greenLED, LOW);


          delay(500);


          if (wrongAttempts >= 3) {
            alarmTriggered = true;
            alertType = "PASSWORD";
          }


          lcd.clear();
          lcd.setCursor(0,0);
          lcd.print("ENTER CODE:");
        }


        input = "";
      }


      else if (key == '*') {
        input = "";
        lcd.setCursor(0,1);
        lcd.print("                ");
      }


      else {
        input += key;


        lcd.setCursor(0,1);
        lcd.print("                ");
        lcd.setCursor(0,1);
        lcd.print(input);
      }
    }


    lastIndex = index;
    delay(120);
  }


  if (index == 0xFF) {
    lastIndex = 0xFF;
  }


  // ================= ALARM =================
  if (alarmTriggered) {


    digitalWrite(greenLED, LOW);


    lcd.clear();
    lcd.setCursor(0,0);
    lcd.print("INTRUDER ALERT!");


    uint8_t alarmIndex = keypad.getKey();


    if (alarmIndex != 0xFF && alarmIndex != lastIndex) {
      if (alarmIndex < 16) {


        char key = keymap[alarmIndex];


        if (key == 'B') {
          secretMode = true;
          input = "";
        }


        else if (key == '#') {


          if (secretMode && input == secretCode) {


            alarmTriggered = false;
            isDisarmed = true;
            wrongAttempts = 0;
            input = "";
            secretMode = false;


            noTone(buzzerPin);


            digitalWrite(redLED, LOW);
            digitalWrite(greenLED, HIGH);


            lcd.clear();
            lcd.setCursor(0,0);
            lcd.print("SYSTEM DISARMED");
          }
          else {
            input = "";
          }
        }


        else if (key == '*') {
          input = "";
        }


        else {
          input += key;
        }
      }


      lastIndex = 0xFF;
      delay(40);
    }


    if (alarmIndex == 0xFF) {
      lastIndex = 0xFF;
    }


    lcd.setCursor(0,1);
    lcd.print("                ");
    lcd.setCursor(0,1);


    if (secretMode) {
      if (input.length() == 0) {
        lcd.print("ENTER SECRET");
      } else {
        for (int i = 0; i < input.length(); i++) {
          lcd.print("*");
        }
      }
    }
    else {
      if (alertType == "MOTION") {
        lcd.print("MOTION DETECTED");
      }
      else if (alertType == "PASSWORD") {
        lcd.print("WRONG PASS x3");
      }
    }


    for (int freq = 800; freq <= 2000; freq += 50) {
      tone(buzzerPin, freq);
      digitalWrite(redLED, HIGH);
      delay(10);
    }


    for (int freq = 2000; freq >= 800; freq -= 50) {
      tone(buzzerPin, freq);
      digitalWrite(redLED, LOW);
      delay(10);
    }
  }
}
