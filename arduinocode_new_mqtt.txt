int irPin = 4;

};

char ssid[] = SECRET_SSID;
char pwd[] = SECRET_PASS;
bool wifi_connected = false;
bool mqtt_connected = false;
int NUM_CONNECTION_TIMES = 4;

const char broker[] = "10.26.216.27"; //replace with real broker
int port = 1883; //Unsecured Port

const char topic1[] = "password_login_time";

String correctCode = "1234";

String input = "";

bool alarmTriggered = false;

void setup() {

  delay(1000);

  Serial.begin(115200);

  // 1. WiFi Connection
      Serial.print("Attempting to connect to SSID: ");
      Serial.println(ssid);

      for (int i = 0; i < NUM_CONNECTION_TIMES; i++) {
          WiFi.begin(ssid, pwd);
          
          // Give it 10 seconds to connect per attempt
          int timeout = 0;
          while (WiFi.status() != WL_CONNECTED && timeout < 20) { 
              delay(500);
              Serial.print(".");
              timeout++;
          }

          if (WiFi.status() == WL_CONNECTED) {
              wifi_connected = true;
              break; 
          }
          Serial.println("\nRetry WiFi...");
      }

      if (wifi_connected) {
          Serial.println("\nConnected to network");
          // NTP Syncing (Your logic here is fine)
          configTime(8 * 3600, 0, "pool.ntp.org");
          struct tm timeinfo;
          while (!getLocalTime(&timeinfo)) {
              delay(500);
          }
      }

      // 2. MQTT Connection
      if (wifi_connected) {
          for (int i = 0; i < NUM_CONNECTION_TIMES; i++) {
              // Many brokers require a Client ID, username, or password
              // If it fails, ensure 'broker' is the IP of your Mosquitto server
              if (mqttClient.connect(broker, port)) {
                  mqtt_connected = true;
                  break;
              }
              delay(1000); // Wait between retries
          }

          if (mqtt_connected) {
              Serial.println("Connected to MQTT broker");
              mqttClient.onMessage(onMqttMessage);
              mqttClient.subscribe(topic2);
              mqttClient.subscribe(topic3);
          } else {
              Serial.print("MQTT connection failed! Error code = ");
              Serial.println(mqttClient.connectError());
          }
      }

//  mqttClient.onMessage(onMqttMessage);

  Wire.begin(21, 22);
  keypad.begin();

  lcd.init();
  lcd.backlight();

  pinMode(redLED, OUTPUT);
  pinMode(greenLED, OUTPUT);
  pinMode(buzzerPin, OUTPUT);
  pinMode(irPin, INPUT_PULLUP);

  digitalWrite(redLED, HIGH);
  digitalWrite(greenLED, LOW);

  lcd.setCursor(0,0);
  lcd.print("ENTER CODE:");

  mySerial.begin(57600, SERIAL_8N1, 16, 17);
  finger.begin(57600);

  if (finger.verifyPassword()) {
    Serial.println("Fingerprint READY");
  } else {
    Serial.println("Fingerprint NOT FOUND");
  }
}

void loop() {

  mqttClient.poll();

  // ================= FINGERPRINT CHECK =================
  if (!isDisarmed && !alarmTriggered) {
    if (finger.getImage() == FINGERPRINT_OK) {
      if (finger.image2Tz() == FINGERPRINT_OK) {
        if (finger.fingerSearch() == FINGERPRINT_OK) {

          control_output_arm_status(true);

          // Publish login success via MQTT
          if (mqtt_connected) {
            struct tm timeinfo;
            char timeStr[30];

            if (getLocalTime(&timeinfo)) {
              strftime(timeStr, sizeof(timeStr), "%Y-%m-%d %H:%M:%S", &timeinfo);
            } else {
              snprintf(timeStr, sizeof(timeStr), "TIME_UNAVAILABLE");
            }

            mqttClient.beginMessage(topic1);
            mqttClient.print("LOGIN_SUCCESS:");
            mqttClient.print(timeStr);
            mqttClient.endMessage();
            mqttClient.beginMessage(topic4);
            mqttClient.print("1"); //indicate successful unlock
            mqttClient.endMessage();
          }

          delay(1000);
          lcd.clear();

        } else {
          lcd.clear();
          lcd.setCursor(0, 0);
          lcd.print("NO MATCH");

          // Low buzz on fingerprint fail
          tone(buzzerPin, 300);
          delay(200);
          noTone(buzzerPin);
          delay(80);
          tone(buzzerPin, 300);
          delay(200);
          noTone(buzzerPin);

          delay(600);
          lcd.clear();
          lcd.setCursor(0, 0);
          lcd.print("ENTER CODE:");
        }
      }
    }
  }

  // ================= DISARMED MODE =================
  if (isDisarmed) {
    reset_arm_status();
    uint8_t index = keypad.getKey();

    if (index != 0xFF) {
      if (index < 16) {
        char key = keymap[index];

        if (key == 'A') {
          control_output_arm_status(false);
          mqttClient.beginMessage(topic4);
            mqttClient.print("0"); //indicate successful lock
            mqttClient.endMessage();
        }
      }

      lastIndex = index;
      delay(150);
    }

    if (index == 0xFF) {
      lastIndex = 0xFF;
    }

    return;
  }

  // ================= MOTION DETECTION =================
  if (digitalRead(irPin) == LOW) {
    alarmTriggered = true;
    alertType = "MOTION";
  }

  // ================= KEYPAD =================
  uint8_t index = keypad.getKey();

  if (index != 0xFF && index != lastIndex && !alarmTriggered) {

    if (index < 16) {
      char key = keymap[index];

      if (key == '#') {
        lcd.clear();

        if (input == correctCode) {
          control_output_arm_status(true);

          if (mqtt_connected) {
            // Sends message to broker signaling time when user unlocked system.
            struct tm timeinfo;
            char timeStr[30];
        
            if (getLocalTime(&timeinfo)) {
                // Formats as e.g. "2025-04-19 14:35:02"
                strftime(timeStr, sizeof(timeStr), "%Y-%m-%d %H:%M:%S", &timeinfo);
            } else {
                snprintf(timeStr, sizeof(timeStr), "TIME_UNAVAILABLE");
            }
        
            mqttClient.beginMessage(topic1);
            mqttClient.print("LOGIN_SUCCESS:");
            mqttClient.print(timeStr);
            mqttClient.endMessage();

            mqttClient.beginMessage(topic4);
            mqttClient.print("1"); //indicate successful unlock
            mqttClient.endMessage();
            
          }
        }
        else {
          wrongAttempts++;

          lcd.setCursor(0,0);
          lcd.print("WRONG PASS");

          digitalWrite(redLED, HIGH);
          digitalWrite(greenLED, LOW);

          delay(500);

          if (wrongAttempts >= 3) {
            alarmTriggered = true;
            alertType = "PASSWORD";
          }

          lcd.clear();
          lcd.setCursor(0,0);
          lcd.print("ENTER CODE:");
        }

        input = "";
      }

      else if (key == '*') {
        input = "";
        lcd.setCursor(0,1);
        lcd.print("                ");
      }

      else {
        input += key;

        lcd.setCursor(0,1);
        lcd.print("                ");
        lcd.setCursor(0,1);
        lcd.print(input);
      }
    }

    lastIndex = index;
    delay(80);
  }

  if (index == 0xFF) {
    lastIndex = 0xFF;
  }

  // ================= ALARM =================
  if (alarmTriggered) {

    digitalWrite(greenLED, LOW);

    lcd.clear();
    lcd.setCursor(0,0);
    lcd.print("INTRUDER ALERT!");

    // 🔥 ADDED (no structure change)
    uint8_t alarmIndex = keypad.getKey();

    if (alarmIndex != 0xFF && alarmIndex != lastIndex) {
      if (alarmIndex < 16) {

        char key = keymap[alarmIndex];

        if (key == 'B') {
          secretMode = true;
          input = "";
        }

        else if (key == '#') {

          if (secretMode && input == secretCode) {
            control_output_arm_status(true);
            mqttClient.beginMessage(topic4);
            mqttClient.print("1"); //indicate successful unlock
            mqttClient.endMessage();
          }

          else {
            input = "";
          }
        }

        else if (key == '*') {
          input = "";
        }

        else {
          input += key;
        }
      }

      lastIndex = 0xFF;
      delay(40);
    }

    if (alarmIndex == 0xFF) {
      lastIndex = 0xFF;
    }

    // 🔥 display (only replaced this part)
    lcd.setCursor(0,1);
    lcd.print("                ");
    lcd.setCursor(0,1);

    if (secretMode) {
      if (input.length() == 0) {
        lcd.print("ENTER SECRET");
      } else {
        for (int i = 0; i < input.length(); i++) {
          lcd.print("*");
        }
      }
    }
    else {
      if (alertType == "MOTION") {
        lcd.print("MOTION DETECTED");
      }
      else if (alertType == "PASSWORD") {
        lcd.print("WRONG PASS x3");
      }
    }

    for (int freq = 800; freq <= 2000; freq += 50) {

          tone(buzzerPin, freq);
          digitalWrite(redLED, HIGH);

          // 🔥 check input multiple times BEFORE delay
          // for (int i = 0; i < 100000000000000000; i++) {
          //   if (keypad.getKey() != 0xFF) break;
          // }

          delay(10); // ✅ keep smooth sound
        }

        for (int freq = 2000; freq >= 800; freq -= 50) {

          tone(buzzerPin, freq);
          digitalWrite(redLED, LOW);

          // // 🔥 check input multiple times BEFORE delay
          // for (int i = 0; i < 10000000000000; i++) {
          //   if (keypad.getKey() != 0xFF) break;
          // }

          delay(10); // ✅ keep smooth sound
        }

  }
}

void reset_arm_status() {
  digitalWrite(redLED, LOW);
  digitalWrite(greenLED, HIGH);
  noTone(buzzerPin);

  lcd.setCursor(0,0);
  lcd.print("SYSTEM DISARMED ");
  lcd.setCursor(0,1);
  lcd.print("Press A to ARM  ");
}

void control_output_arm_status(bool disarm_state) {
  if (disarm_state == true) {
    isDisarmed = true;
    wrongAttempts = 0;
    secretMode = false; // 🔥 reset

    lcd.setCursor(0,0);
    lcd.print("DISARMED");

    digitalWrite(redLED, LOW);
    digitalWrite(greenLED, HIGH);

    tone(buzzerPin, 1200);
    delay(150);
    noTone(buzzerPin);
    delay(100);
    tone(buzzerPin, 1500);
    delay(150);
    noTone(buzzerPin);

    lcd.clear();
    reset_arm_status();
  }
  else {
    isDisarmed = false;
    alarmTriggered = false;
    wrongAttempts = 0;
    input = "";
    secretMode = false; // 🔥 reset
  
    lcd.clear();
    lcd.setCursor(0,0);
    lcd.print("SYSTEM ARMED");
  
    digitalWrite(redLED, HIGH);
    digitalWrite(greenLED, LOW);
  
    tone(buzzerPin, 800);
    delay(150);
    noTone(buzzerPin);
  
    delay(800);
  
    lcd.clear();
    lcd.setCursor(0,0);
    lcd.print("ENTER CODE:");
  }
}

void handle_password_setting_logic(byte* payload, unsigned int len) {
    // ✅ Parse the incoming payload as the new password string
    String newPassword = "";
    for (unsigned int i = 0; i < len; i++) {
        newPassword += (char)payload[i];
    }

    if (newPassword.length() > 0) {
        correctCode = newPassword;
        Serial.print("Password updated to: ");
        Serial.println(correctCode);
    }
}

void handle_update_arm_status_logic(byte* payload, unsigned int len) {
    if (len == 0) return;

    String payloadStr = "";
    for (unsigned int i = 0; i < len; i++) {
        payloadStr += (char)payload[i];
    }

    if (payloadStr == "1" || payloadStr == "true") {
        control_output_arm_status(true);   // Disarm
    } else if (payloadStr == "0" || payloadStr == "false") {
        control_output_arm_status(false);  // Arm
    }
}
void onMqttMessage(int messageSize) {
    String topic = mqttClient.messageTopic();
    
    // Read payload into byte array
    unsigned int len = messageSize;
    byte payload[len];
    for (unsigned int i = 0; i < len; i++) {
        payload[i] = mqttClient.read();
    }

    // If topic is website setting password.
    if (topic == topic2) {
        handle_password_setting_logic(payload, len);
    }
    else if (topic == topic3) {
      handle_update_arm_status_logic(payload, len);
    }
}
