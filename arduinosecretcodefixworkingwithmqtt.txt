#include <Wire.h>
#include <I2CKeyPad.h>
#include <LiquidCrystal_I2C.h>
#include <ArduinoMqttClient.h>
#include <WiFi.h>
#include <string.h>
#include <time.h>
#include <Adafruit_Fingerprint.h>
#include <HardwareSerial.h>

#define KEYPAD_ADDR 0x20
#define LCD_ADDR 0x27

// Replace with own credentials (@Kok Ho)
#define SECRET_SSID "OnePlus 12R"
#define SECRET_PASS "michellengai"

I2CKeyPad keypad(KEYPAD_ADDR);
LiquidCrystal_I2C lcd(LCD_ADDR, 16, 2);

// Fingerprint setup
HardwareSerial mySerial(2);
Adafruit_Fingerprint finger = Adafruit_Fingerprint(&mySerial);

WiFiClient wifiClient;
MqttClient mqttClient(wifiClient);

// LED pins
int redLED = 3;
int greenLED = 5;

// IR sensor
int irPin = 4;

// buzzer
int buzzerPin = 18;

// keypad mapping
char keymap[16] = {
  'D', '#', '0', '*',
  'C', '9', '8', '7',
  'B', '6', '5', '4',
  'A', '3', '2', '1'
};

uint8_t lastIndex = 0xFF;

char ssid[] = SECRET_SSID;
char pwd[] = SECRET_PASS;
bool wifi_connected = false;
bool mqtt_connected = false;
int NUM_CONNECTION_TIMES = 4;

const char broker[] = "10.26.216.27";
int port = 1883;

const char topic1[] = "password_login_time";
const char topic2[] = "set_login_password";

const long interval = 8000;
unsigned long previousMillis = 0;

// password system
String correctCode = "1234";
String secretCode = "6969";
String input = "";
int wrongAttempts = 0;

bool isDisarmed = false;
bool alarmTriggered = false;
bool secretMode = false;
String alertType = "";


void setup() {

  delay(1000);

  Serial.begin(115200);

  // 1. WiFi Connection
  Serial.print("Attempting to connect to SSID: ");
  Serial.println(ssid);

  for (int i = 0; i < NUM_CONNECTION_TIMES; i++) {
    WiFi.begin(ssid, pwd);

    int timeout = 0;
    while (WiFi.status() != WL_CONNECTED && timeout < 20) {
      delay(500);
      Serial.print(".");
      timeout++;
    }

    if (WiFi.status() == WL_CONNECTED) {
      wifi_connected = true;
      break;
    }
    Serial.println("\nRetry WiFi...");
  }

  if (wifi_connected) {
    Serial.println("\nConnected to network");
    configTime(8 * 3600, 0, "pool.ntp.org");
    struct tm timeinfo;
    while (!getLocalTime(&timeinfo)) {
      delay(500);
    }
  }

  // 2. MQTT Connection
  if (wifi_connected) {
    for (int i = 0; i < NUM_CONNECTION_TIMES; i++) {
      if (mqttClient.connect(broker, port)) {
        mqtt_connected = true;
        break;
      }
      delay(1000);
    }

    if (mqtt_connected) {
      Serial.println("Connected to MQTT broker");
      mqttClient.onMessage(onMqttMessage);
      mqttClient.subscribe(topic2);
    } else {
      Serial.print("MQTT connection failed! Error code = ");
      Serial.println(mqttClient.connectError());
    }
  }

  mqttClient.onMessage(onMqttMessage);

  Wire.begin(21, 22);
  keypad.begin();

  lcd.init();
  lcd.backlight();

  pinMode(redLED, OUTPUT);
  pinMode(greenLED, OUTPUT);
  pinMode(buzzerPin, OUTPUT);
  pinMode(irPin, INPUT_PULLUP);

  digitalWrite(redLED, HIGH);
  digitalWrite(greenLED, LOW);

  lcd.setCursor(0, 0);
  lcd.print("ENTER CODE:");

  // Fingerprint init
  mySerial.begin(57600, SERIAL_8N1, 16, 17);
  finger.begin(57600);

  if (finger.verifyPassword()) {
    Serial.println("Fingerprint READY");
  } else {
    Serial.println("Fingerprint NOT FOUND");
  }
}


void loop() {

  // Keep MQTT alive
  mqttClient.poll();

  // ================= FINGERPRINT CHECK =================
  if (!isDisarmed && !alarmTriggered) {
    if (finger.getImage() == FINGERPRINT_OK) {
      if (finger.image2Tz() == FINGERPRINT_OK) {
        if (finger.fingerSearch() == FINGERPRINT_OK) {

          // MATCH → DISARM
          isDisarmed = true;
          alarmTriggered = false;
          wrongAttempts = 0;
          input = "";
          secretMode = false;

          digitalWrite(redLED, LOW);
          digitalWrite(greenLED, HIGH);

          // Success beeps — same as keypad disarm
          tone(buzzerPin, 1200);
          delay(150);
          noTone(buzzerPin);
          delay(100);
          tone(buzzerPin, 1500);
          delay(150);
          noTone(buzzerPin);

          lcd.clear();
          lcd.setCursor(0, 0);
          lcd.print("FINGERPRINT OK");

          // Publish login success via MQTT
          if (mqtt_connected) {
            struct tm timeinfo;
            char timeStr[30];

            if (getLocalTime(&timeinfo)) {
              strftime(timeStr, sizeof(timeStr), "%Y-%m-%d %H:%M:%S", &timeinfo);
            } else {
              snprintf(timeStr, sizeof(timeStr), "TIME_UNAVAILABLE");
            }

            mqttClient.beginMessage(topic1);
            mqttClient.print("LOGIN_SUCCESS:");
            mqttClient.print(timeStr);
            mqttClient.endMessage();
          }

          delay(1000);
          lcd.clear();

        } else {
          lcd.clear();
          lcd.setCursor(0, 0);
          lcd.print("NO MATCH");

          // Low buzz on fingerprint fail
          tone(buzzerPin, 300);
          delay(200);
          noTone(buzzerPin);
          delay(80);
          tone(buzzerPin, 300);
          delay(200);
          noTone(buzzerPin);

          delay(600);
          lcd.clear();
          lcd.setCursor(0, 0);
          lcd.print("ENTER CODE:");
        }
      }
    }
  }

  // ================= DISARMED MODE =================
  if (isDisarmed) {

    digitalWrite(redLED, LOW);
    digitalWrite(greenLED, HIGH);
    noTone(buzzerPin);

    lcd.setCursor(0, 0);
    lcd.print("SYSTEM DISARMED ");
    lcd.setCursor(0, 1);
    lcd.print("Press A to ARM  ");

    uint8_t index = keypad.getKey();

    if (index != 0xFF) {
      if (index < 16) {
        char key = keymap[index];

        if (key == 'A') {
          isDisarmed = false;
          alarmTriggered = false;
          wrongAttempts = 0;
          input = "";
          secretMode = false;

          lcd.clear();
          lcd.setCursor(0, 0);
          lcd.print("SYSTEM ARMED");

          digitalWrite(redLED, HIGH);
          digitalWrite(greenLED, LOW);

          tone(buzzerPin, 800);
          delay(150);
          noTone(buzzerPin);

          delay(800);

          lcd.clear();
          lcd.setCursor(0, 0);
          lcd.print("ENTER CODE:");
        }
      }

      lastIndex = index;
      delay(150);
    }

    if (index == 0xFF) {
      lastIndex = 0xFF;
    }

    return;
  }

  // ================= MOTION DETECTION =================
  if (digitalRead(irPin) == LOW) {
    alarmTriggered = true;
    alertType = "MOTION";
  }

  // ================= KEYPAD =================
  uint8_t index = keypad.getKey();

  if (index != 0xFF && index != lastIndex && !alarmTriggered) {

    if (index < 16) {
      char key = keymap[index];

      if (key == '#') {
        lcd.clear();

        if (input == correctCode) {

          isDisarmed = true;
          wrongAttempts = 0;
          secretMode = false;

          lcd.setCursor(0, 0);
          lcd.print("DISARMED");

          digitalWrite(redLED, LOW);
          digitalWrite(greenLED, HIGH);

          tone(buzzerPin, 1200);
          delay(150);
          noTone(buzzerPin);
          delay(100);
          tone(buzzerPin, 1500);
          delay(150);
          noTone(buzzerPin);

          // Publish login success via MQTT
          if (mqtt_connected) {
            struct tm timeinfo;
            char timeStr[30];

            if (getLocalTime(&timeinfo)) {
              strftime(timeStr, sizeof(timeStr), "%Y-%m-%d %H:%M:%S", &timeinfo);
            } else {
              snprintf(timeStr, sizeof(timeStr), "TIME_UNAVAILABLE");
            }

            mqttClient.beginMessage(topic1);
            mqttClient.print("LOGIN_SUCCESS:");
            mqttClient.print(timeStr);
            mqttClient.endMessage();
          }

        } else {
          wrongAttempts++;

          lcd.setCursor(0, 0);
          lcd.print("WRONG PASS");

          digitalWrite(redLED, HIGH);
          digitalWrite(greenLED, LOW);

          // Wrong password beep — low double buzz
          tone(buzzerPin, 300);
          delay(200);
          noTone(buzzerPin);
          delay(80);
          tone(buzzerPin, 300);
          delay(200);
          noTone(buzzerPin);

          delay(500);

          if (wrongAttempts >= 3) {
            alarmTriggered = true;
            alertType = "PASSWORD";
          }

          lcd.clear();
          lcd.setCursor(0, 0);
          lcd.print("ENTER CODE:");
        }

        input = "";
      }

      else if (key == '*') {
        // Clear — short low tick
        tone(buzzerPin, 600);
        delay(60);
        noTone(buzzerPin);

        input = "";
        lcd.setCursor(0, 1);
        lcd.print("                ");
      }

      else {
        // Keyclick — short neutral tick per digit
        tone(buzzerPin, 1000);
        delay(60);
        noTone(buzzerPin);

        input += key;

        lcd.setCursor(0, 1);
        lcd.print("                ");
        lcd.setCursor(0, 1);
        lcd.print(input);
      }
    }

    lastIndex = index;
    delay(80);
  }

  if (index == 0xFF) {
    lastIndex = 0xFF;
  }

  // ================= ALARM =================
  if (alarmTriggered) {

    digitalWrite(greenLED, LOW);

    // If NOT in secret mode, run the siren sweep and show alert text
    if (!secretMode) {
      lcd.clear();
      lcd.setCursor(0, 0);
      lcd.print("INTRUDER ALERT!");
      lcd.setCursor(0, 1);
      lcd.print("                ");
      lcd.setCursor(0, 1);
      if (alertType == "MOTION") {
        lcd.print("MOTION DETECTED");
      } else if (alertType == "PASSWORD") {
        lcd.print("WRONG PASS x3");
      }

      // Siren sweep — checks for B key to enter secret mode mid-sweep
      for (int freq = 800; freq <= 2000; freq += 50) {
        tone(buzzerPin, freq);
        digitalWrite(redLED, HIGH);
        uint8_t k = keypad.getKey();
        if (k != 0xFF && k < 16 && keymap[k] == 'B') {
          secretMode = true;
          input = "";
          lastIndex = 0xFF;
          break;
        }
        delay(10);
      }

      if (!secretMode) {
        for (int freq = 2000; freq >= 800; freq -= 50) {
          tone(buzzerPin, freq);
          digitalWrite(redLED, LOW);
          uint8_t k = keypad.getKey();
          if (k != 0xFF && k < 16 && keymap[k] == 'B') {
            secretMode = true;
            input = "";
            lastIndex = 0xFF;
            break;
          }
          delay(10);
        }
      }
    }

    // If in secret mode — siren keeps alerting, show input prompt, handle keys
    if (secretMode) {

      lcd.clear();
      lcd.setCursor(0, 0);
      lcd.print("ENTER SECRET:");
      lcd.setCursor(0, 1);
      lcd.print("                ");
      lcd.setCursor(0, 1);
      if (input.length() == 0) {
        lcd.print("_");
      } else {
        for (int i = 0; i < (int)input.length(); i++) lcd.print("*");
      }

      // Run a short siren burst while waiting for keypress
      // Each step is 10ms — total ~480ms per half sweep, enough to sound alarming
      // but loop stays responsive to keypresses
      for (int freq = 800; freq <= 2000; freq += 50) {
        tone(buzzerPin, freq);
        digitalWrite(redLED, HIGH);
        delay(8);
      }
      for (int freq = 2000; freq >= 800; freq -= 50) {
        tone(buzzerPin, freq);
        digitalWrite(redLED, LOW);
        delay(8);
      }
      noTone(buzzerPin);

      uint8_t alarmIndex = keypad.getKey();

      if (alarmIndex != 0xFF && alarmIndex != lastIndex && alarmIndex < 16) {
        char key = keymap[alarmIndex];

        if (key == '#') {
          if (input == secretCode) {
            // ✅ Correct secret — disarm fully
            alarmTriggered = false;
            isDisarmed = true;
            wrongAttempts = 0;
            input = "";
            secretMode = false;

            noTone(buzzerPin);
            digitalWrite(redLED, LOW);
            digitalWrite(greenLED, HIGH);

            tone(buzzerPin, 1200);
            delay(150);
            noTone(buzzerPin);
            delay(100);
            tone(buzzerPin, 1500);
            delay(150);
            noTone(buzzerPin);

            lcd.clear();
            lcd.setCursor(0, 0);
            lcd.print("SYSTEM DISARMED");
          } else {
            // ❌ Wrong secret — back to full alarm, clear everything
            input = "";
            secretMode = false;
          }
        }

        else if (key == '*') {
          // Cancel — resume alarm
          input = "";
          secretMode = false;
        }

        else {
          // Digit keyclick — brief high beep over the siren gap
          tone(buzzerPin, 1000);
          delay(60);
          noTone(buzzerPin);
          input += key;
        }

        lastIndex = alarmIndex;
        delay(80);
      }

      if (alarmIndex == 0xFF) {
        lastIndex = 0xFF;
      }
    }
  }
}


void handle_password_setting_logic(byte* payload, unsigned int len) {
  String newPassword = "";
  for (unsigned int i = 0; i < len; i++) {
    newPassword += (char)payload[i];
  }

  if (newPassword.length() > 0) {
    correctCode = newPassword;
    Serial.print("Password updated to: ");
    Serial.println(correctCode);
  }
}


void onMqttMessage(int messageSize) {
  String topic = mqttClient.messageTopic();

  unsigned int len = messageSize;
  byte payload[len];
  for (unsigned int i = 0; i < len; i++) {
    payload[i] = mqttClient.read();
  }

  if (topic == topic2) {
    handle_password_setting_logic(payload, len);
  }
}
